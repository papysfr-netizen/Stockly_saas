rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function estConnecte() {
      return request.auth != null;
    }

    function profilExiste() {
      return exists(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    function monProfil() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    function monEntreprise() {
      return monProfil().entrepriseId;
    }

    function monRole() {
      return monProfil().get('role', 'caissier');
    }

    function estActif() {
      return monProfil().get('estActif', true) != false;
    }

    function estMembre(entId) {
      return estConnecte() && profilExiste() && monEntreprise() == entId && estActif();
    }

    function estProprietaireDe(entId) {
      return estConnecte()
        && get(/databases/$(database)/documents/entreprises/$(entId)).data.ownerId == request.auth.uid;
    }

    function aRoleGerantOuPlus() {
      return monRole() in ['owner', 'proprietaire', 'patron', 'gerant', 'admin',
                           'administrateur', 'chefboutique', 'chef_boutique', 'chef', 'comptable'];
    }

    function estGerantOuPlus(entId) {
      return estMembre(entId) && (estProprietaireDe(entId) || aRoleGerantOuPlus());
    }

    function aPermission(entId, cle) {
      return estMembre(entId) && (
        estProprietaireDe(entId)
        || monProfil().get('permissions', {}).get(cle, aRoleGerantOuPlus()) == true
      );
    }

    function creeParMoi(entId) {
      return estConnecte()
        && getAfter(/databases/$(database)/documents/entreprises/$(entId)).data.ownerId == request.auth.uid;
    }

    function maBoutique() {
      return monProfil().get('boutiqueId', '') == '' ? 'principale' : monProfil().get('boutiqueId', '');
    }

    function aToutesBoutiques(entId) {
      return estMembre(entId)
        && monProfil().get('permissions', {}).get('toutesBoutiques', false) == true;
    }

    function aParametres(entId) {
      return estMembre(entId)
        && monProfil().get('permissions', {}).get('parametresSensibles', false) == true;
    }

    function boutiqueAutorisee(entId, b) {
      return estProprietaireDe(entId)
        || aToutesBoutiques(entId)
        || (b == '' ? 'principale' : b) == maBoutique();
    }

    function paiementValide() {
      let avant = resource.data;
      let apres = request.resource.data;
      let resteAvant = avant.get('montantReste', avant.get('montantRestant', avant.get('montantInitial', 0)));
      let baisse = resteAvant - apres.get('montantReste', 0);
      let paye = apres.get('acomptePaye', 0) - avant.get('acomptePaye', 0);
      let rendu = apres.get('montantRetour', 0) - avant.get('montantRetour', 0);
      return apres.diff(avant).affectedKeys().hasOnly(['montantReste', 'montantRestant', 'acomptePaye',
                 'statut', 'dernierPaiement', 'updatedAt', 'paiements', 'montantRetour'])
        && apres.get('montantReste', 0) >= 0
        && baisse >= 0
        && paye >= 0
        && rendu >= 0
        && baisse <= paye + rendu + 0.02;
    }

    function venteIntacte() {
      return !request.resource.data.diff(resource.data).affectedKeys().hasAny([
        'id', 'entrepriseId', 'montantTotal', 'articles', 'remiseTotale',
        'vendeurId', 'vendeurNom', 'boutiqueId', 'date', 'createdAt',
        'nomClient', 'telephoneClient', 'methodePaiement', 'tauxEchange',
        'montantPayeUSD', 'montantPayeLocal', 'montantCompteClient',
        'monnaieRendueUSD', 'monnaieRendueLocal', 'comptabilise', 'coutMarchandises',
        'prixVendus', 'monnaieDevise', 'monnaieDonneeUSD', 'monnaieDonneeCDF'
      ]);
    }

    function lignesCoherentes(v) {
      return verifLignes(v.articles, v.prixVendus, v.montantTotal);
    }

    function verifLignes(a, p, total) {
      return a is list && a.size() <= 30 && p is map
        && total >= (
             (a.size() <= 0 ? 0 : p[a[0].produitId] * a[0].quantite) + (a.size() <= 1 ? 0 : p[a[1].produitId] * a[1].quantite) + (a.size() <= 2 ? 0 : p[a[2].produitId] * a[2].quantite)
           + (a.size() <= 3 ? 0 : p[a[3].produitId] * a[3].quantite) + (a.size() <= 4 ? 0 : p[a[4].produitId] * a[4].quantite) + (a.size() <= 5 ? 0 : p[a[5].produitId] * a[5].quantite)
           + (a.size() <= 6 ? 0 : p[a[6].produitId] * a[6].quantite) + (a.size() <= 7 ? 0 : p[a[7].produitId] * a[7].quantite) + (a.size() <= 8 ? 0 : p[a[8].produitId] * a[8].quantite)
           + (a.size() <= 9 ? 0 : p[a[9].produitId] * a[9].quantite) + (a.size() <= 10 ? 0 : p[a[10].produitId] * a[10].quantite) + (a.size() <= 11 ? 0 : p[a[11].produitId] * a[11].quantite)
           + (a.size() <= 12 ? 0 : p[a[12].produitId] * a[12].quantite) + (a.size() <= 13 ? 0 : p[a[13].produitId] * a[13].quantite) + (a.size() <= 14 ? 0 : p[a[14].produitId] * a[14].quantite)
           + (a.size() <= 15 ? 0 : p[a[15].produitId] * a[15].quantite) + (a.size() <= 16 ? 0 : p[a[16].produitId] * a[16].quantite) + (a.size() <= 17 ? 0 : p[a[17].produitId] * a[17].quantite)
           + (a.size() <= 18 ? 0 : p[a[18].produitId] * a[18].quantite) + (a.size() <= 19 ? 0 : p[a[19].produitId] * a[19].quantite) + (a.size() <= 20 ? 0 : p[a[20].produitId] * a[20].quantite)
           + (a.size() <= 21 ? 0 : p[a[21].produitId] * a[21].quantite) + (a.size() <= 22 ? 0 : p[a[22].produitId] * a[22].quantite) + (a.size() <= 23 ? 0 : p[a[23].produitId] * a[23].quantite)
           + (a.size() <= 24 ? 0 : p[a[24].produitId] * a[24].quantite) + (a.size() <= 25 ? 0 : p[a[25].produitId] * a[25].quantite) + (a.size() <= 26 ? 0 : p[a[26].produitId] * a[26].quantite)
           + (a.size() <= 27 ? 0 : p[a[27].produitId] * a[27].quantite) + (a.size() <= 28 ? 0 : p[a[28].produitId] * a[28].quantite) + (a.size() <= 29 ? 0 : p[a[29].produitId] * a[29].quantite)) - 0.05;
    }

    function nomClientValide(nom) {
      return nom is string && nom.trim().size() > 0
        && nom.trim().lower() != 'client divers';
    }

    function telephoneValide(tel) {
      return tel is string && tel.replace('[^0-9]', '').size() >= 9;
    }

    function enSuppression(entId) {
      return get(/databases/$(database)/documents/entreprises/$(entId)).data.get('enSuppression', false) == true;
    }

    function estAdminEquipe(entId) {
      return estMembre(entId)
        && monProfil().get('permissions', {}).get('gererEquipe', false) == true;
    }

    function profilEstAdmin(data) {
      return data.get('permissions', {}).get('gererEquipe', false) == true;
    }

    match /users/{userId} {

      allow read: if estConnecte() && (
        request.auth.uid == userId
        || estProprietaireDe(resource.data.entrepriseId)
        || estAdminEquipe(resource.data.entrepriseId)
      );

      allow create: if estConnecte() && (
        (request.auth.uid == userId && creeParMoi(request.resource.data.entrepriseId))
        || (
          estProprietaireDe(request.resource.data.entrepriseId)
          && request.resource.data.role in ['gerant', 'caissier']
        )

        || (
          estAdminEquipe(request.resource.data.entrepriseId)
          && request.resource.data.role in ['gerant', 'caissier']
          && request.resource.data.get('uid', '') == userId
          && request.resource.data.get('estActif', true) == true
          && request.resource.data.get('permissions', {}).keys().hasOnly(['canGiveDiscount'])
          && request.resource.data.get('permissions', {}).get('canGiveDiscount', false) == false
          && !request.resource.data.keys().hasAny(['boutiqueId'])
        )
      );

      allow update: if estConnecte() && (
        (

          request.auth.uid == userId
          && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['role', 'entrepriseId', 'uid', 'estActif', 'permissions', 'boutiqueId'])
        )
        || (
          estProprietaireDe(resource.data.entrepriseId)
          && resource.data.role != 'owner'
          && request.resource.data.entrepriseId == resource.data.entrepriseId
          && request.resource.data.role in ['gerant', 'caissier']
        )

        || (
          estAdminEquipe(resource.data.entrepriseId)
          && request.auth.uid != userId
          && resource.data.role in ['gerant', 'caissier']
          && request.resource.data.role in ['gerant', 'caissier']
          && !profilEstAdmin(resource.data)
          && get(/databases/$(database)/documents/entreprises/$(resource.data.entrepriseId)).data.ownerId != userId
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['role', 'estActif'])
        )
      );

      allow delete: if estConnecte() && (
        (request.auth.uid == userId
          && !exists(/databases/$(database)/documents/entreprises/$(resource.data.entrepriseId)))
        || (estProprietaireDe(resource.data.entrepriseId) && enSuppression(resource.data.entrepriseId))

        || (request.auth.uid != userId
            && resource.data.get('role', '') in ['gerant', 'caissier']
            && (estProprietaireDe(resource.data.entrepriseId)
                || (estAdminEquipe(resource.data.entrepriseId) && !profilEstAdmin(resource.data))))
      );
    }

    match /entreprises/{entId} {
      allow read: if estMembre(entId);
      allow create: if estConnecte() && request.resource.data.ownerId == request.auth.uid;

      allow update: if estProprietaireDe(entId) && request.resource.data.ownerId == resource.data.ownerId;

      allow delete: if estProprietaireDe(entId) && resource.data.get('enSuppression', false) == true;

      match /{partie=**} {
        allow delete: if estProprietaireDe(entId) && enSuppression(entId);
      }

      match /produits/{produitId} {
        allow read: if estMembre(entId);

        allow create, delete: if aPermission(entId, 'gererStock');

        allow update: if majProduit();

        function majProduit() {
          let profil = get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
          let perms = profil.get('permissions', {});
          let gerant = profil.get('role', 'caissier') in ['owner', 'proprietaire', 'patron', 'gerant', 'admin',
                           'administrateur', 'chefboutique', 'chef_boutique', 'chef', 'comptable'];
          let proprio = get(/databases/$(database)/documents/entreprises/$(entId)).data.ownerId == request.auth.uid;
          let b = profil.get('boutiqueId', '') == '' ? 'principale' : profil.get('boutiqueId', '');
          let apres = request.resource.data;
          let ecart = apres.get('quantite', 0) - resource.data.get('quantite', 0);
          return profil.get('entrepriseId', '') == entId
            && profil.get('estActif', true) != false
            && (

              proprio || perms.get('gererStock', gerant) == true

              || (apres.diff(resource.data).affectedKeys().hasOnly(['quantite', 'stocks', 'updatedAt', 'derniereVente', 'lotsVendus', 'lotsCA'])
                  && (perms.get('toutesBoutiques', false) == true
                      || (apres.get('stocks', {}).diff(resource.data.get('stocks', {})).affectedKeys().hasOnly([b])

                          && (b == 'principale'
                              || apres.get('stocks', {}).get(b, 0) - resource.data.get('stocks', {}).get(b, 0) == ecart)))

                  && (ecart <= 0 || perms.get('retours', gerant) == true)

                  && (ecart >= 0 || perms.get('remisesRetraits', gerant) == true || venduAuBonPrix())));
        }

        function venduAuBonPrix() {
          let venteId = request.resource.data.get('derniereVente', '');
          let chemin = /databases/$(database)/documents/entreprises/$(entId)/ventes/$(venteId);
          let vente = getAfter(chemin).data;
          let b = vente.get('boutiqueId', '');
          let prixBoutique = resource.data.get('prixParBoutique', {}).get(b, 0);
          let normal = prixBoutique > 0 ? prixBoutique : resource.data.get('prixVente', 0);
          return venteId != ''
            && !exists(chemin)
            && vente.get('prixVendus', {}).get(produitId, -1) >= normal - 0.01;
        }

      }

      match /boutiques/{boutiqueId} {
        allow read: if estMembre(entId);

        allow create, update: if estProprietaireDe(entId) || aParametres(entId);
        allow delete: if false;
      }

      match /transferts/{transfertId} {
        allow read: if estMembre(entId);
        allow create: if aPermission(entId, 'gererStock');
        allow update, delete: if false;
      }

      match /ventes/{docId} {
        allow read: if estMembre(entId);

        allow update: if estGerantOuPlus(entId)
          || (estMembre(entId) && venteIntacte()

              && request.resource.data.get('montantRetourne', 0) >= resource.data.get('montantRetourne', 0)
              && request.resource.data.get('montantAnnule', 0) >= resource.data.get('montantAnnule', 0))
          || (aPermission(entId, 'maintenance')
              && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['coutMarchandises']));

        allow create: if estMembre(entId)
          && boutiqueAutorisee(entId, request.resource.data.get('boutiqueId', ''))
          && (estGerantOuPlus(entId)
              || request.resource.data.get('vendeurId', '') == request.auth.uid)
          && ((request.resource.data.get('remiseTotale', 0) <= 0.01
               && lignesCoherentes(request.resource.data))
              || aPermission(entId, 'remisesRetraits'))

          && (request.resource.data.get('resteAPayer', 0) <= 0.01
              || (telephoneValide(request.resource.data.get('telephoneClient', ''))
                  && nomClientValide(request.resource.data.get('nomClient', ''))));
        allow delete: if estGerantOuPlus(entId);
      }
      match /retours/{docId} {

        allow read: if estMembre(entId);
        allow create: if aPermission(entId, 'retours') && retourCoherent();
        allow update, delete: if false;

        function retourCoherent() {
          let r = request.resource.data;
          let chemin = /databases/$(database)/documents/entreprises/$(entId)/ventes/$(r.get('venteId', ''));
          let avant = get(chemin).data;
          let apres = getAfter(chemin).data;
          let deja = avant.get('montantRetourne', 0) + avant.get('montantAnnule', 0);
          let ensuite = apres.get('montantRetourne', 0) + apres.get('montantAnnule', 0);
          return math.abs(ensuite - deja - r.get('montant', 0)) <= 0.01
            && ensuite <= avant.get('montantTotal', 0) + 0.01
            && r.get('montantRembourse', 0) + r.get('reductionDette', 0) + r.get('montantConserve', 0)
               <= r.get('montant', 0) + 0.01;
        }
      }
      match /comptes_clients/{compteId} {

        allow read: if estMembre(entId);

        allow create: if estMembre(entId) && (estProprietaireDe(entId) || soldeJustifie(0));
        allow update: if estMembre(entId) && (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['totalRetire'])
          || aPermission(entId, 'remisesRetraits')
        ) && (estProprietaireDe(entId) || soldeJustifie(resource.data.get('solde', 0)));
        allow delete: if false;

        function soldeJustifie(soldeAvant) {
          let ecart = request.resource.data.get('solde', 0) - soldeAvant;
          let mvtId = request.resource.data.get('dernierMouvementId', '');
          let chemin = /databases/$(database)/documents/entreprises/$(entId)/comptes_clients/$(compteId)/mouvements/$(mvtId);
          return math.abs(ecart) <= 0.005
            || (mvtId != '' && !exists(chemin)
                && math.abs(getAfter(chemin).data.montant * getAfter(chemin).data.sens - ecart) <= 0.01);
        }

        match /mouvements/{mvtId} {
          allow read: if estMembre(entId);

          allow create: if estMembre(entId)
            && request.resource.data.get('type', '') in ['depot', 'avoir', 'achat', 'retrait']
            && request.resource.data.get('montant', -1) >= 0
            && (request.resource.data.get('sens', 0) == -1
                || (request.resource.data.get('sens', 0) == 1
                    && request.resource.data.type in ['depot', 'avoir']))
            && (request.resource.data.type != 'retrait' || aPermission(entId, 'remisesRetraits'))
            && (request.resource.data.type != 'avoir' || avoirJustifie());
          allow update, delete: if false;

          function avoirJustifie() {
            let retourId = request.resource.data.get('retourId', '');
            let chemin = /databases/$(database)/documents/entreprises/$(entId)/retours/$(retourId);
            let r = getAfter(chemin).data;
            return retourId != '' && !exists(chemin)
              && r.get('modeRemboursement', '') == 'avoir'
              && r.get('compteClient', '') == compteId
              && math.abs(r.get('montantRembourse', 0) - request.resource.data.montant) <= 0.01;
          }
        }
      }

      match /creances/{docId} {
        allow read: if estMembre(entId);

        allow create: if estMembre(entId)
          && telephoneValide(request.resource.data.get('telephone', ''))
          && nomClientValide(request.resource.data.get('nomTiers', ''));
        allow update: if estGerantOuPlus(entId) || (estMembre(entId) && paiementValide());
        allow delete: if estGerantOuPlus(entId);
      }
      match /dettes/{docId} {
        allow read, create: if estMembre(entId);
        allow update: if estGerantOuPlus(entId) || (estMembre(entId) && paiementValide());
        allow delete: if estGerantOuPlus(entId);
      }

      match /ecritures_comptables/{docId} {
        allow read: if aPermission(entId, 'voirFinances');
        allow create: if estMembre(entId);
        allow update, delete: if estGerantOuPlus(entId);
      }

      match /arrivages/{arrivageId} {
        allow read: if estMembre(entId);
        allow create: if aPermission(entId, 'gererStock');
        allow update: if aPermission(entId, 'gererStock')
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['statut', 'annuleLe', 'annulePar'])
          && resource.data.get('statut', 'actif') == 'actif'
          && request.resource.data.statut == 'annule';
        allow delete: if false;
      }

      match /soldes_mensuels/{mois} {
        allow read, write: if aPermission(entId, 'voirFinances');
      }

      match /clotures/{clotureId} {
        allow read: if estMembre(entId);
        allow create: if estMembre(entId)
          && boutiqueAutorisee(entId, request.resource.data.boutiqueId)
          && clotureId == request.resource.data.boutiqueId + '_' + request.resource.data.jour
          && request.resource.data.jour.matches('[0-9]{4}-[0-9]{2}-[0-9]{2}')
          && request.resource.data.parUid == request.auth.uid
          && request.resource.data.createdAt == request.time
          && request.resource.data.keys().hasOnly(['boutiqueId', 'jour', 'parUid', 'parNom', 'taux',
               'fondUSD', 'fondCDF', 'compteUSD', 'compteCDF', 'compteMobile',
               'fondDemainUSD', 'fondDemainCDF', 'note', 'createdAt'])
          && request.resource.data.compteUSD >= 0
          && request.resource.data.compteCDF >= 0
          && request.resource.data.compteMobile >= 0
          && request.resource.data.fondUSD >= 0
          && request.resource.data.fondCDF >= 0
          && request.resource.data.note.size() <= 300;
        allow update: if false;
        allow delete: if estGerantOuPlus(entId);
      }

      match /journal/{docId} {
        allow read: if estGerantOuPlus(entId) || aParametres(entId);
        allow create: if estMembre(entId) && request.resource.data.userId == request.auth.uid;
        allow update, delete: if false;
      }

      match /{collection}/{reste=**} {
        allow read: if estMembre(entId) && !(collection in ['journal', 'ecritures_comptables', 'soldes_mensuels']);
        allow write: if !(collection in ['produits', 'ventes', 'retours', 'comptes_clients', 'boutiques', 'transferts', 'creances', 'dettes', 'ecritures_comptables', 'journal', 'clotures', 'soldes_mensuels', 'arrivages'])
          && (estMembre(entId) || creeParMoi(entId));
      }
    }

    match /depenses/{depenseId} {
      allow read: if estGerantOuPlus(monEntreprise()) && resource.data.entrepriseId == monEntreprise();
      allow create: if estGerantOuPlus(monEntreprise()) && request.resource.data.entrepriseId == monEntreprise()
        && boutiqueAutorisee(monEntreprise(), request.resource.data.get('boutiqueId', ''));
      allow update: if estGerantOuPlus(monEntreprise()) && resource.data.entrepriseId == monEntreprise()
        && request.resource.data.entrepriseId == resource.data.entrepriseId;
      allow delete: if estGerantOuPlus(monEntreprise()) && resource.data.entrepriseId == monEntreprise();
    }

    match /rapports_erreurs/{rapportId} {
      allow create: if estConnecte()
        && request.resource.data.keys().hasOnly(['uid', 'message', 'pile', 'contexte', 'fatale', 'plateforme', 'date'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.message is string && request.resource.data.message.size() <= 1000
        && request.resource.data.pile is string && request.resource.data.pile.size() <= 4000
        && request.resource.data.contexte is string && request.resource.data.contexte.size() <= 200
        && request.resource.data.date == request.time;
      allow read, update, delete: if false;
    }

    match /organizations/{entId}/{document=**} {
      allow read, write: if false;
    }
  }
}
